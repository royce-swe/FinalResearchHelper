import React, { useState } from "react";

interface FacultyMember {
  id: string;
  name: string;
  email: string;
  department?: string;
  research_area?: string;
}

interface FacultyFinderProps {
  onProfessorSelect: (professorId: string) => void;
}

const FacultyFinder: React.FC<FacultyFinderProps> = ({ onProfessorSelect }) => {
  const [userMajor, setUserMajor] = useState("");
  const [userSchool, setUserSchool] = useState("");
  const [facultyMembers, setFacultyMembers] = useState<FacultyMember[]>([]);
  const [displayCount, setDisplayCount] = useState(3);
  const [isLoading, setIsLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState("");

  // Fetch professors dynamically from your backend
  const fetchProfessors = async (
    school: string,
    major: string
  ): Promise<FacultyMember[] | null> => {
    try {
      const response = await fetch("http://127.0.0.1:5050/find-professors", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ university: school, interests: [major] }),
      });

      if (!response.ok) throw new Error("Server error while fetching data.");

      const data = await response.json();

      // Transform professor data
      const faculty = (data.professors || []).map((prof: any) => ({
        id: prof.id,
        name: prof.name,
        email: prof.email || "Not Available",
        department: prof.research_area || "N/A",
      }));

      return faculty;
    } catch (error) {
      console.error("Fetch failed:", error);
      setErrorMessage("Could not fetch professor data. Please try again.");
      return null;
    }
  };

  // Handle search button click
  const handleSubmit = async () => {
    if (!userSchool || !userMajor) {
      setErrorMessage("Please enter both a school and a field of study.");
      return;
    }

    setIsLoading(true);
    setErrorMessage("");
    setFacultyMembers([]);
    setDisplayCount(3);

    const faculty = await fetchProfessors(userSchool, userMajor);

    if (faculty === null || faculty.length === 0) {
      setFacultyMembers([]);
      setErrorMessage("No matching professors found. Try adjusting your search.");
    } else {
      setFacultyMembers(faculty);
    }

    setIsLoading(false);
  };

  // Load more professors (show 3 more)
  const handleLoadMore = () => {
    setDisplayCount((prev) => prev + 3);
  };

  // Slice the displayed professors based on displayCount
  const visibleProfessors = facultyMembers.slice(0, displayCount);

  return (
    <div className="min-h-screen bg-gradient-to-br from-blue-50 via-white to-purple-50 pt-20">
      <div className="max-w-4xl mx-auto px-4 sm:px-6 lg:px-8 py-16">
        <div className="text-center mb-12">
          <h1 className="text-5xl font-bold text-gray-900 mb-4">Faculty Finder</h1>
          <p className="text-xl text-gray-600">
            Find professors by university and field of study
          </p>
          <p className="text-sm text-gray-500 italic mt-1">
            <strong>Note:</strong> The search may take a couple of minutes to complete.
          </p>
        </div>

        <div className="glass rounded-3xl shadow-2xl p-8 md:p-12">
          <div className="space-y-8">
            {/* Field of Study */}
            <div>
              <label className="block text-2xl font-semibold text-gray-900 mb-4">
                What's your field of study?
              </label>
              <input
                type="text"
                value={userMajor}
                onChange={(e) => setUserMajor(e.target.value)}
                placeholder="e.g., Machine Learning"
                className="w-full px-6 py-4 text-lg border-2 border-gray-200 rounded-xl focus:border-blue-500 focus:outline-none transition-colors"
              />
              {userMajor && (
                <div className="mt-3 p-3 bg-blue-50 rounded-lg border-l-4 border-blue-500">
                  <p className="text-lg text-blue-600 font-medium">Field: {userMajor}</p>
                </div>
              )}
            </div>

            {/* University Input */}
            <div>
              <label className="block text-2xl font-semibold text-gray-900 mb-4">
                Select your target university
              </label>
              <input
                type="text"
                value={userSchool}
                onChange={(e) => setUserSchool(e.target.value)}
                placeholder="e.g., Stanford University"
                className="w-full px-6 py-4 text-lg border-2 border-gray-200 rounded-xl focus:border-purple-500 focus:outline-none transition-colors"
              />
              {userSchool && (
                <div className="mt-3 p-3 bg-purple-50 rounded-lg border-l-4 border-purple-500">
                  <p className="text-lg text-purple-600 font-medium">University: {userSchool}</p>
                </div>
              )}
            </div>

            {errorMessage && (
              <div className="bg-red-100 text-red-700 px-4 py-3 rounded-md">
                {errorMessage}
              </div>
            )}

            <button
              onClick={handleSubmit}
              disabled={isLoading}
              className="w-full btn-primary text-xl py-4 disabled:opacity-50 disabled:cursor-not-allowed disabled:transform-none"
            >
              {isLoading ? (
                <>
                  <span
                    className="animate-spin inline-block w-6 h-6 border-4 border-blue-600 border-t-transparent rounded-full mr-3"
                    aria-label="loading"
                  ></span>
                  Finding Professors...
                </>
              ) : (
                "Search →"
              )}
            </button>
          </div>

          {visibleProfessors.length > 0 && (
            <div className="mt-12 p-8 bg-gradient-to-r from-gray-50 to-blue-50 rounded-2xl border border-gray-200">
              <h3 className="text-2xl font-bold text-gray-900 mb-6 text-center flex items-center justify-center">
                <span className="mr-2">📧</span>
                Professors Found
              </h3>
              <div className="space-y-4">
                {visibleProfessors.map((faculty) => {
                  const hasValidEmail =
                    faculty.email &&
                    faculty.email !== "Not Available" &&
                    faculty.email !== "N/A";
                  const displayEmail = hasValidEmail ? faculty.email : "N/A";
                  return (
                    <div
                      key={faculty.id}
                      className="bg-white p-6 rounded-lg shadow-sm border-l-4 border-blue-500 hover:shadow-md transition-shadow duration-200"
                    >
                      <div className="flex items-center justify-between">
                        <div className="flex items-center flex-1">
                          <span className="text-2xl mr-4">👨‍🏫</span>
                          <div>
                            <h4 className="text-lg font-semibold text-gray-900">
                              {faculty.name}
                            </h4>
                            <p className="text-gray-600">{displayEmail}</p>
                            {faculty.department && (
                              <p className="text-sm text-gray-500">{faculty.department}</p>
                            )}
                          </div>
                        </div>
                        <div className="flex space-x-3">
                          <a
                            href={hasValidEmail ? `mailto:${faculty.email}` : "#"}
                            className={`px-4 py-2 rounded-lg font-medium transition-colors ${
                              hasValidEmail
                                ? "bg-gray-100 text-gray-700 hover:bg-gray-200"
                                : "bg-gray-300 text-gray-400 cursor-not-allowed"
                            }`}
                            onClick={(e) => {
                              if (!hasValidEmail) e.preventDefault();
                            }}
                            tabIndex={hasValidEmail ? 0 : -1}
                            aria-disabled={!hasValidEmail}
                          >
                            Quick Email
                          </a>
                          <button
                            onClick={() => onProfessorSelect(faculty.id)}
                            className="px-4 py-2 bg-blue-600 text-white rounded-lg font-medium hover:bg-blue-700 transition-colors"
                          >
                            Learn More →
                          </button>
                        </div>
                      </div>
                    </div>
                  );
                })}
              </div>

              {/* Load More button */}
              {displayCount < facultyMembers.length && (
                <button
                  onClick={handleLoadMore}
                  disabled={isLoading}
                  className="w-full mt-6 bg-white text-gray-700 py-3 px-6 rounded-lg font-semibold hover:bg-gray-50 transition-colors border-2 border-gray-200 hover:border-blue-300 disabled:opacity-50"
                >
                  Load More Professors
                </button>
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

export default FacultyFinder;
